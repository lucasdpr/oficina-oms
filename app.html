const isCadeiraInf = tipoUpper.includes('CADEIRA INFERIOR'); const isSegmento = tipoUpper.includes('SEGMENTO') || tipoUpper.includes('SEGUIMENTO'); if (isBow) { placeholder = 'Bow'; for (let i = 1; i <= 5;
